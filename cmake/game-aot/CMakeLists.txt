cmake_minimum_required(VERSION 3.20)
project(YakumoActualGameCorpus LANGUAGES CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../RequirePrivateBuild.cmake")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(YAKUMO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(generated "${YAKUMO_ROOT}/out/data/game-boot/generated")
if(NOT EXISTS "${generated}/generated_registry.cpp")
    message(FATAL_ERROR "Prepare the verified local executable and original automatic corpus first")
endif()
set(sources "${generated}/generated_registry.cpp")
foreach(unit RANGE 0 88)
    string(LENGTH "${unit}" digits)
    math(EXPR zeros "4 - ${digits}")
    string(REPEAT "0" ${zeros} padding)
    list(APPEND sources "${generated}/generated_unit_${padding}${unit}.cpp")
endforeach()
add_library(actual_game_aot STATIC ${sources})
target_include_directories(actual_game_aot PRIVATE "${YAKUMO_ROOT}/upstream/Yakumo/include" "${generated}")
# Keep the original ABI gate reproducible; optimized candidates use a new build directory.
set(YAKUMO_AOT_OPTIMIZATION "0" CACHE STRING "Game corpus optimization: 0 or 2")
if(NOT YAKUMO_AOT_OPTIMIZATION MATCHES "^(0|2)$")
    message(FATAL_ERROR "Only the diagnostic O0 and measured O2 variants are supported")
endif()
target_compile_options(actual_game_aot PRIVATE -O${YAKUMO_AOT_OPTIMIZATION} -g -fexceptions -frtti -ffunction-sections -fdata-sections)
