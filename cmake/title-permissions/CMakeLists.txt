cmake_minimum_required(VERSION 3.22)
project(yakumo_title_permissions C ASM)
include("${CMAKE_CURRENT_LIST_DIR}/../RequirePrivateBuild.cmake")
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
option(YAKUMO_NATIVE "Build the manual PS5 permission payload" OFF)
option(YAKUMO_PERMISSION_DIAGNOSTICS "Instrument the frozen repair and SDK initialization" OFF)
set(YAKUMO_PERMISSION_SOURCE_OVERRIDE "" CACHE FILEPATH "Optional identified permission payload source")
set(YAKUMO_PERMISSION_CRT_OVERRIDE "" CACHE FILEPATH "Optional identified permission payload CRT")
set(YAKUMO_PERMISSION_OUTPUT_NAME "payload.elf" CACHE STRING "Permission payload output filename")
if(YAKUMO_NATIVE AND (NOT YAKUMO_PERMISSION_SOURCE_OVERRIDE OR NOT YAKUMO_PERMISSION_CRT_OVERRIDE))
    message(FATAL_ERROR "Prepare the existing final2 source/CRT with tools/prepare_payload_sources.py and pass both explicit overrides; do not build the historical base policy as a replacement")
endif()
set(PERMISSION_SOURCE "${ROOT}/src/title_permissions.c")
if(YAKUMO_PERMISSION_SOURCE_OVERRIDE)
    set(PERMISSION_SOURCE "${YAKUMO_PERMISSION_SOURCE_OVERRIDE}")
endif()
if(YAKUMO_PERMISSION_DIAGNOSTICS)
    set(diag_args)
    if(YAKUMO_NATIVE)
        list(APPEND diag_args --native)
    endif()
    execute_process(COMMAND python3 "${ROOT}/tools/project_permission_diagnostics.py"
        --output "${CMAKE_CURRENT_BINARY_DIR}" ${diag_args} COMMAND_ERROR_IS_FATAL ANY)
    if(NOT YAKUMO_PERMISSION_SOURCE_OVERRIDE)
        set(PERMISSION_SOURCE "${CMAKE_CURRENT_BINARY_DIR}/title_permissions_diagnostic.c")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
        "${ROOT}/tools/project_permission_diagnostics.py" "${ROOT}/src/title_permissions.c"
        "${ROOT}/src/permission_bootstrap_status.h")
endif()
add_executable(title_permissions "${PERMISSION_SOURCE}")
target_include_directories(title_permissions PRIVATE "${ROOT}/src")
target_compile_features(title_permissions PRIVATE c_std_11)
target_compile_options(title_permissions PRIVATE -Wall -Wextra -Werror -g -O1)
target_compile_definitions(title_permissions PRIVATE YAKUMO_NATIVE=$<BOOL:${YAKUMO_NATIVE}>)
if(YAKUMO_NATIVE)
    set(SDK "${ROOT}/deps/PS5_PayloadSDK")
    if(YAKUMO_PERMISSION_CRT_OVERRIDE)
        set(CRT "${YAKUMO_PERMISSION_CRT_OVERRIDE}")
        set_source_files_properties("${CRT}" PROPERTIES COMPILE_OPTIONS "-ffreestanding;-fno-builtin")
    elseif(YAKUMO_PERMISSION_DIAGNOSTICS)
        set(CRT "${CMAKE_CURRENT_BINARY_DIR}/crt-diagnostic.c")
        # This executes before dynamic libc imports are relocated. The ordinary
        # optimization of its BSS loop into memset would call an unresolved GOT.
        set_source_files_properties("${CRT}" PROPERTIES COMPILE_OPTIONS "-ffreestanding;-fno-builtin")
    else()
        set(CRT "${CMAKE_CURRENT_BINARY_DIR}/crt-return.c")
        execute_process(COMMAND python3 "${ROOT}/tools/prepare_installation_crt.py" "${CRT}" COMMAND_ERROR_IS_FATAL ANY)
    endif()
    target_sources(title_permissions PRIVATE "${CRT}")
    foreach(unit syscall klog nid kernel rtld rtld_so rtld_sprx rtld_payload rtld_dlfcn mdbg patch kernel_procio)
        target_sources(title_permissions PRIVATE "${SDK}/crt/${unit}.c")
        set_source_files_properties("${SDK}/crt/${unit}.c" PROPERTIES COMPILE_OPTIONS "-ffreestanding;-fno-builtin;-Wno-unused-parameter;-Wno-sign-compare")
    endforeach()
    target_sources(title_permissions PRIVATE "${SDK}/crt/_start.S")
    target_include_directories(title_permissions PRIVATE "${SDK}/crt" "${ROOT}/src")
    target_compile_options(title_permissions PRIVATE -fPIC -ffunction-sections -fdata-sections -fms-extensions -Wno-microsoft-anon-tag)
    target_link_options(title_permissions PRIVATE -nostartfiles -Wl,--gc-sections "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/payload.map")
    get_filename_component(payload_name "${YAKUMO_PERMISSION_OUTPUT_NAME}" NAME_WE)
    get_filename_component(payload_suffix "${YAKUMO_PERMISSION_OUTPUT_NAME}" LAST_EXT)
    set_target_properties(title_permissions PROPERTIES OUTPUT_NAME "${payload_name}" SUFFIX "${payload_suffix}")
else()
    enable_testing()
    if(YAKUMO_PERMISSION_DIAGNOSTICS)
        add_executable(permission_bootstrap_tests "${ROOT}/tests/permission_bootstrap_tests.c")
        target_include_directories(permission_bootstrap_tests PRIVATE "${ROOT}/src" "${ROOT}/deps/PS5_PayloadSDK/crt")
        target_compile_options(permission_bootstrap_tests PRIVATE -Wall -Wextra -Werror)
        add_test(NAME permission_bootstrap_tests COMMAND permission_bootstrap_tests)
    endif()
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_library(permission_faults SHARED "${ROOT}/tests/title_permissions_faults.c")
    target_link_libraries(permission_faults PRIVATE dl)
    target_compile_options(permission_faults PRIVATE -Wall -Wextra -Werror)
    add_test(NAME title_permissions_tests COMMAND ${Python3_EXECUTABLE} "${ROOT}/tests/title_permissions_tests.py" $<TARGET_FILE:title_permissions> $<TARGET_FILE:permission_faults>)
    if(YAKUMO_PERMISSION_SOURCE_OVERRIDE)
        set_tests_properties(title_permissions_tests PROPERTIES ENVIRONMENT "MHP3RD_PERMISSION_SUCCESS=Permissions applied;MHP3RD_PERMISSION_CHANGE=changes=")
    endif()
endif()
