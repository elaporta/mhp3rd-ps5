cmake_minimum_required(VERSION 3.20)
project(YakumoWebSettings LANGUAGES CXX)
get_filename_component(ONLINE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)

set(YAKUMO_PROJECTION_SCRIPT "${ONLINE_ROOT}/tools/project_web_settings_sources.py" CACHE FILEPATH "" FORCE)
set(YAKUMO_GAME_ENTRY "${CMAKE_CURRENT_BINARY_DIR}/projection/online_entry.cpp" CACHE FILEPATH "" FORCE)
add_compile_definitions(YAKUMO_GAME_WEB=1)
include("${ONLINE_ROOT}/cmake/first-frame/CMakeLists.txt")
target_sources(first_frame_profile PRIVATE "${ONLINE_ROOT}/src/adhoc_resolver.cpp" "${ONLINE_ROOT}/src/game_network.cpp"
    "${ONLINE_ROOT}/src/adhoc_log.cpp" "${ONLINE_ROOT}/src/game_web.cpp")
if(CMAKE_CROSSCOMPILING)
    target_sources(first_frame_profile PRIVATE "${ONLINE_ROOT}/src/adhoc_socket.cpp")
endif()
include("${ONLINE_ROOT}/cmake/online-tests.cmake")
if(NOT CMAKE_CROSSCOMPILING)
    add_executable(game_web_entropy_tests "${ONLINE_ROOT}/tests/game_web_entropy_tests.cpp")
    target_include_directories(game_web_entropy_tests PRIVATE "${ONLINE_ROOT}/src")
    add_test(NAME web_settings_kernel_entropy COMMAND game_web_entropy_tests)
    add_executable(game_web_tests "${ONLINE_ROOT}/tests/game_web_tests.cpp")
    target_compile_definitions(game_web_tests PRIVATE WEB_ASSETS="${ONLINE_ROOT}/assets/web")
    target_link_libraries(game_web_tests PRIVATE first_frame_profile)
    target_link_options(game_web_tests PRIVATE -Wl,--gc-sections)
    add_test(NAME web_settings_http_and_lifecycle COMMAND game_web_tests)
    set_tests_properties(web_settings_http_and_lifecycle PROPERTIES TIMEOUT 45)
    add_executable(game_web_native_tests "${ONLINE_ROOT}/tests/game_web_tests.cpp"
        "${ONLINE_ROOT}/tests/game_web_native_fixture.cpp" "${ONLINE_ROOT}/src/game_web.cpp")
    target_compile_definitions(game_web_native_tests PRIVATE YAKUMO_NATIVE=1 WEB_ASSETS="${ONLINE_ROOT}/assets/web")
    target_link_libraries(game_web_native_tests PRIVATE first_frame_profile)
    target_link_options(game_web_native_tests PRIVATE -Wl,--gc-sections)
    add_test(NAME web_settings_native_listener_and_browser COMMAND game_web_native_tests)
    set_tests_properties(web_settings_native_listener_and_browser PROPERTIES TIMEOUT 45 RUN_SERIAL TRUE)
    add_executable(game_web_import_absence_tests "${ONLINE_ROOT}/tests/game_web_tests.cpp"
        "${ONLINE_ROOT}/tests/game_web_native_fixture.cpp" "${ONLINE_ROOT}/src/game_web.cpp")
    target_compile_definitions(game_web_import_absence_tests PRIVATE YAKUMO_NATIVE=1 YAKUMO_WEB_HOST_WEAK_FIXTURE=1 WEB_ASSETS="${ONLINE_ROOT}/assets/web")
    target_link_libraries(game_web_import_absence_tests PRIVATE first_frame_profile)
    target_link_options(game_web_import_absence_tests PRIVATE -Wl,--gc-sections)
    add_test(NAME web_settings_unbound_native_imports COMMAND game_web_import_absence_tests)
    set_tests_properties(web_settings_unbound_native_imports PROPERTIES TIMEOUT 15 RUN_SERIAL TRUE)
    set_tests_properties(web_settings_http_and_lifecycle PROPERTIES RUN_SERIAL TRUE)
    if(YAKUMO_TEXT_COVERAGE)
        set_property(SOURCE "${ONLINE_ROOT}/src/game_web.cpp" APPEND PROPERTY COMPILE_OPTIONS -fprofile-instr-generate -fcoverage-mapping)
        foreach(target game_web_tests game_web_native_tests game_web_import_absence_tests game_web_entropy_tests)
            target_compile_options(${target} PRIVATE -fprofile-instr-generate -fcoverage-mapping)
            target_link_options(${target} PRIVATE -fprofile-instr-generate)
        endforeach()
    endif()
endif()
